# chrombpnet-tensorflow-glibc.yml's pip section: requirements-tensorflow.txt's
# stack at releases with glibc 2.17 builds.
#
# The newest releases of some dependencies ship Linux wheels only for glibc
# 2.27 or newer, plus source code, which pip would try to compile;
# --prefer-binary has it take the newest release with a wheel for this
# machine instead. The pins below keep their versions; it decides the rest.
--prefer-binary

# TensorFlow 2.20.0 is the last release whose wheels run on glibc 2.17 (2.21
# needs 2.27), with its CUDA 12.5 wheels (manylinux2014 builds; any driver >=
# 525). tf-keras 2.20 is its Keras 2 (the tensorflow engine); Keras 3.15.1,
# pure Python, is the tensorflow-keras3 engine's, as in chrombpnet-tensorflow.
tensorflow[and-cuda]==2.20.0
tf-keras==2.20.1
keras==3.15.1
tensorflow-probability==0.25.0
# cuDNN 9.3, the release TF 2.20 is built against: newer 9.x drop the
# precompiled kernels for Pascal and Volta GPUs (as in requirements-tensorflow.txt).
nvidia-cudnn-cu12==9.3.0.75
# TensorFlow's; 0.5.3 and newer need glibc 2.27.
ml-dtypes==0.5.1

# DeepSHAP's dinucleotide shuffle (the tensorflow engine's DeepSHAP is copied
# into chrombpnet/engines/tensorflow_old/deep_tf.py).
deeplift==0.6.13.0
weasyprint>=61

# PyPI has igraph 1.0.0 and leidenalg 0.11.0 (modisco pins it) for glibc 2.28
# only: build-glibc217-wheels.sh builds them (igraph without GraphML file
# support, which needs libxml2), and PIP_FIND_LINKS points pip at its output.
# On a glibc 2.17 host pip can take only those; on a newer one it prefers
# PyPI's, so reinstall them from the wheel directory with --no-index.
igraph==1.0.0
leidenalg==0.11.0
# 0.68's wheels need glibc 2.28.
numba==0.67.0
# modisco's. PyPI's wheel works, but its optional HTJ2K filter needs glibc
# 2.27 and warns on import; a source build (in the same wheel directory) has
# none, and pip takes it only with --no-index.
hdf5plugin==7.1.0

# The commit chrombpnet-pytorch-glibc has.
modisco @ git+https://github.com/kundajelab/tfmodisco.git@48e0f87e08bcc6bff973d6cd6b08ad6a94f3eb51
pytest>=8
pytest-cov>=5
